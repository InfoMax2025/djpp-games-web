var ASSETS = [
  './',
  './index.html',
  './manifest.webmanifest',
  './icon-192.png',
  './icon-512.png',
  './icon-maskable-512.png',
  './favicon.png',
  './icon-apple-180.png'
];

self.addEventListener('install', function(e){
  self.skipWaiting();
  e.waitUntil(caches.open(CACHE).then(function(c){ return c.addAll(ASSETS); }).catch(function(){}));
});

self.addEventListener('activate', function(e){
  e.waitUntil(
    caches.keys().then(function(keys){
      return Promise.all(keys.map(function(k){ if(k!==CACHE) return caches.delete(k); }));
    }).then(function(){ return self.clients.claim(); })
  );
});

self.addEventListener('fetch', function(e){
  if(e.request.method !== 'GET') return;
  e.respondWith(
    caches.match(e.request).then(function(hit){
      if(hit) return hit;
      return fetch(e.request).then(function(res){
        var copy = res.clone();
        caches.open(CACHE).then(function(c){ c.put(e.request, copy); }).catch(function(){});
        return res;
      }).catch(function(){
        return caches.match('./index.html');
      });
    })
  );
});

(function(){
'use strict';
// ---------- constants ----------
var SUITS=['H','D','C','S'];
var SUIT_SYM={H:'♥',D:'♦',C:'♣',S:'♠'};
var SUIT_NAME={H:'Hearts',D:'Diamonds',C:'Clubs',S:'Spades'};
var SUIT_COLOR={H:'red',D:'red',C:'blk',S:'blk'};
var RANKS=['9','10','J','Q','K','A'];
var RANK_VAL={'9':9,'10':10,'J':11,'Q':12,'K':13,'A':14};
// seats: 0 South(you) 1 West 2 North(partner) 3 East
var SEATNAME=['You','West','Partner','East'];
var SEAT_TAG=['S','W','N','E'];
var TEAM=[0,1,0,1]; // 0 = you+partner, 1 = opponents

function sameColor(a,b){ return SUIT_COLOR[a]===SUIT_COLOR[b]; }
function leftSuit(trump){ // suit of the left bower
  for(var i=0;i<SUITS.length;i++) if(SUITS[i]!==trump && sameColor(SUITS[i],trump)) return SUITS[i];
}

// ---------- state ----------
var G=null; // game
function newGame(){
  G={ scores:[0,0], dealer:Math.floor(Math.random()*4), inProgress:true };
  el('scoreUs').textContent=0; el('scoreThem').textContent=0;
}

// ---------- deck ----------
function makeDeck(){
  var d=[];
  for(var s=0;s<4;s++) for(var r=0;r<6;r++) d.push({s:SUITS[s],r:RANKS[r]});
  return d;
}
function shuffle(a){
  for(var i=a.length-1;i>0;i--){ var j=Math.floor(Math.random()*(i+1)); var t=a[i];a[i]=a[j];a[j]=t; }
  return a;
}

// ---------- card power under trump ----------
function isRight(c,trump){ return c.r==='J' && c.s===trump; }
function isLeft(c,trump){ return c.r==='J' && c.s===leftSuit(trump); }
function effSuit(c,trump){ // suit a card "belongs to" for following
  if(isLeft(c,trump)) return trump;
  return c.s;
}
function cardPower(c,trump,led){
  // higher = stronger, in context of led suit
  if(isRight(c,trump)) return 1000;
  if(isLeft(c,trump)) return 900;
  if(c.s===trump) return 500+RANK_VAL[c.r];
  if(led && effSuit(c,trump)===led) return 100+RANK_VAL[c.r];
  return RANK_VAL[c.r]; // off-suit, can't win
}

// ---------- deal ----------
function deal(){
  var deck=shuffle(makeDeck());
  var hands=[[],[],[],[]];
  // deal 5 each (simple 5x4)
  for(var k=0;k<5;k++) for(var p=0;p<4;p++) hands[p].push(deck.pop());
  var up=deck.pop();
  G.hands=hands; G.up=up; G.kitty=deck; // remaining 3 unused
  G.trump=null; G.maker=-1; G.alone=-1; G.caller=-1;
  G.tricks=[0,0,0,0]; G.trickCount=0;
  G.leader=(G.dealer+1)%4;
  sortHand(0);
}
function sortHand(p){
  // stable-ish display sort by suit then rank (trump-aware once trump known)
  var t=G.trump;
  G.hands[p].sort(function(a,b){
    var sa=t?effSuit(a,t):a.s, sb=t?effSuit(b,t):b.s;
    if(sa!==sb) return SUITS.indexOf(sa)-SUITS.indexOf(sb);
    if(t) return cardPower(b,t,null)-cardPower(a,t,null);
    return RANK_VAL[b.r]-RANK_VAL[a.r];
  });
}

// ---------- bidding ----------
var STICK_DEALER=true; // dealer must call in round 2

function handStrength(hand, trump, considerUp){
  // rough 0..~40 heuristic
  var s=0;
  for(var i=0;i<hand.length;i++){
    var c=hand[i];
    if(isRight(c,trump)) s+=15;
    else if(isLeft(c,trump)) s+=13;
    else if(c.s===trump) s+=6+RANK_VAL[c.r]/5;
    else if(c.r==='A') s+=3; // off ace
    else s+=0.4;
  }
  return s;
}

// ---------- AI bidding decisions ----------
function aiBidRound1(p){
  // upcard suit is candidate trump; dealer would pick it up
  var trump=G.up.s;
  var hyp=G.hands[p].slice();
  var strength=handStrength(hyp, trump, true);
  // dealer gets the upcard (bonus); dealer's team benefit
  var dealerTeam = TEAM[G.dealer]===TEAM[p];
  if(dealerTeam) strength+=3;
  if(p===G.dealer) strength+=3;
  var alone = strength>28;
  if(strength>=17) return {call:true, suit:trump, alone:alone};
  return {call:false};
}
function aiBidRound2(p){
  var best=null;
  for(var i=0;i<SUITS.length;i++){
    var suit=SUITS[i];
    if(suit===G.up.s) continue; // can't name turned-down suit
    var strength=handStrength(G.hands[p], suit, false);
    if(!best || strength>best.strength) best={suit:suit,strength:strength};
  }
  var mustCall = STICK_DEALER && p===G.dealer;
  var alone = best.strength>28;
  if(best.strength>=17 || mustCall) return {call:true, suit:best.suit, alone:alone};
  return {call:false};
}

// ---------- setting trump ----------
function setTrump(suit, maker, alone){
  G.trump=suit; G.maker=maker; G.caller=maker; G.alone=alone?maker:-1;
  el('trumpSym').textContent=SUIT_SYM[suit];
  el('trumpSym').style.color=SUIT_COLOR[suit]==='red'?'#ff6b6b':'#e8e8e8';
  markCaller(maker);
}
function dealerPickup(){
  // dealer picks up upcard and discards worst
  var h=G.hands[G.dealer];
  h.push(G.up);
  // discard lowest-power card (not trump if avoidable)
  var worst=0, wp=1e9;
  for(var i=0;i<h.length;i++){
    var pw=cardPower(h[i],G.trump,null);
    if(pw<wp){ wp=pw; worst=i; }
  }
  h.splice(worst,1);
  if(G.dealer===0) sortHand(0);
}

// ---------- trick play ----------
function legalMoves(p){
  var h=G.hands[p], t=G.trump, led=G.ledSuit;
  if(led==null) return h.map(function(_,i){return i;});
  var canFollow=[];
  for(var i=0;i<h.length;i++) if(effSuit(h[i],t)===led) canFollow.push(i);
  if(canFollow.length) return canFollow;
  return h.map(function(_,i){return i;});
}
function aiPlay(p){
  var h=G.hands[p], t=G.trump, led=G.ledSuit;
  var legal=legalMoves(p);
  // current winning card in trick
  var win=currentTrickWinner();
  var partnerWinning = win.seat>=0 && TEAM[win.seat]===TEAM[p];
  // map legal to cards with power
  var opts=legal.map(function(i){ return {i:i, c:h[i], pw:cardPower(h[i],t,led)}; });
  if(led==null){
    // leading: lead a strong trump early if we're maker, else lead off-ace, else lowest
    opts.sort(function(a,b){return b.pw-a.pw;});
    // prefer leading an off-suit ace (power 100+14) or right bower; avoid wasting
    var ace=opts.find(function(o){return o.c.r==='A' && effSuit(o.c,t)!==t;});
    if(TEAM[p]===TEAM[G.maker] && opts[0].pw>=900) return opts[0].i; // lead bower to draw trump
    if(ace) return ace.i;
    // lead middling
    return opts[Math.floor(opts.length/2)].i;
  }
  // following
  var winners=opts.filter(function(o){return o.pw>win.power;});
  if(partnerWinning && win.power>=500){
    // partner has a strong trump/led winner — throw lowest
    opts.sort(function(a,b){return a.pw-b.pw;});
    return opts[0].i;
  }
  if(winners.length){
    // win as cheaply as possible
    winners.sort(function(a,b){return a.pw-b.pw;});
    return winners[0].i;
  }
  // can't win — throw lowest
  opts.sort(function(a,b){return a.pw-b.pw;});
  return opts[0].i;
}
function currentTrickWinner(){
  var best={seat:-1,power:-1,card:null};
  for(var i=0;i<G.play.length;i++){
    var pl=G.play[i];
    var pw=cardPower(pl.card,G.trump,G.ledSuit);
    if(pw>best.power){ best={seat:pl.seat,power:pw,card:pl.card}; }
  }
  return best;
}

// ================= FLOW =================
var busy=false;

function bidStep(){

  setActive(p);

  if(decision.call){

  advanceBid();
}
function advanceBid(){
  G.passes++;
  if(G.bidRound===1 && G.passes>=4){
    // flip upcard down, round 2
    G.bidRound=2; G.passes=0; G.bidTurn=(G.dealer+1)%4;
    el('upcardFace').classList.add('turned');
    toast('Up-card turned down');
    setTimeout(bidStep,650);
    return;
  }
  if(G.bidRound===2 && G.passes>=4){
    // all passed (only if stick-dealer off) -> redeal
    toast('Everyone passed — redeal');
    G.dealer=(G.dealer+1)%4;
    setTimeout(startHand,900);
    return;
  }
  G.bidTurn=(G.bidTurn+1)%4;
  setTimeout(bidStep,600);
}
function commitCall(p, suit, alone, round1){

function doPlay(p, idx){
  if(G.resolving) return;                 // ignore stray inputs during resolution
  if(p!==G.turn) return;                   // only the active seat may play
  if(idx<0 || idx>=G.hands[p].length) return;
  var card=G.hands[p].splice(idx,1)[0];
  if(G.ledSuit==null) G.ledSuit=effSuit(card,G.trump);

// deterministic-ish auto play to validate engine (no rendering dependency issues)
function autoGame(maxHands){
  // Run engine headlessly by simulating decisions synchronously.
  var log={hands:0, err:null, finalScores:null};
  try{
    var localDealer=Math.floor(Math.random()*4);
    var scores=[0,0];
    var guard=0;
    while(scores[0]<10 && scores[1]<10 && guard<200){
      guard++;
      var r=simOneHand(localDealer);
      scores[0]+=r.pts[0]; scores[1]+=r.pts[1];
      log.hands++;
      localDealer=(localDealer+1)%4;
      if(r.err){ log.err=r.err; break; }
    }
    log.finalScores=scores;
  }catch(e){ log.err=String(e)+' @ '+(e.stack||'').split('\n')[1]; }

}
function simOneHand(dealer){
  var deck=shuffle(makeDeck());
  var hands=[[],[],[],[]];
  for(var k=0;k<5;k++) for(var p=0;p<4;p++) hands[p].push(deck.pop());
  var up=deck.pop();
  var S={hands:hands,up:up,dealer:dealer,trump:null,maker:-1,alone:-1,tricks:[0,0,0,0]};
  // bidding r1
  var maker=-1,trump=null,alone=false,round1=false;
  var order=[(dealer+1)%4,(dealer+2)%4,(dealer+3)%4,dealer];
  for(var i=0;i<4;i++){
    var p=order[i]; var strength=hStr(hands[p],up.s);
    if(TEAM[dealer]===TEAM[p]) strength+=3; if(p===dealer) strength+=3;
    if(strength>=17){ maker=p; trump=up.s; alone=strength>28; round1=true; break; }
  }
  if(maker<0){
    for(var i2=0;i2<4;i2++){
      var p2=order[i2]; var best=null;
      for(var s2=0;s2<4;s2++){ if(SUITS[s2]===up.s) continue; var st=hStr(hands[p2],SUITS[s2]); if(!best||st>best.st) best={su:SUITS[s2],st:st}; }
      var must=STICK_DEALER&&p2===dealer;
      if(best.st>=17||must){ maker=p2; trump=best.su; alone=best.st>28; break; }
    }
  }
  if(maker<0){ return {pts:[0,0], err:null, redeal:true}; }
  S.trump=trump; S.maker=maker; S.alone=alone?maker:-1;
  if(round1){ // dealer picks up
    hands[dealer].push(up);
    var worst=0,wp=1e9; for(var d=0;d<hands[dealer].length;d++){ var pw=cardPow(hands[dealer][d],trump,null); if(pw<wp){wp=pw;worst=d;} }
    hands[dealer].splice(worst,1);
  }
  var sitOut = alone?(maker+2)%4:-1;
  var leader=(dealer+1)%4; if(leader===sitOut) leader=(leader+1)%4;
  // verify hand sizes
  for(var v=0;v<4;v++){ if(v===dealer){ if(hands[v].length!==5) return {pts:[0,0],err:'dealer hand '+hands[v].length}; } else if(hands[v].length!==5) return {pts:[0,0],err:'hand '+v+' size '+hands[v].length}; }
  // play 5 tricks
  for(var tr=0;tr<5;tr++){
    var play=[]; var led=null; var turn=leader; var cnt=sitOut>=0?3:4;
    for(var c=0;c<cnt;c++){
      if(turn===sitOut) turn=(turn+1)%4;
      var legal=legalSim(hands[turn],trump,led);
      if(legal.length===0) return {pts:[0,0],err:'no legal moves seat '+turn};
      var idx=aiPlaySim(hands[turn],trump,led,play,turn);
      if(legal.indexOf(idx)<0) return {pts:[0,0],err:'illegal ai move seat '+turn};
      var card=hands[turn].splice(idx,1)[0];
      if(led==null) led=effSuit(card,trump);
      play.push({seat:turn,card:card});
      turn=(turn+1)%4; if(turn===sitOut) turn=(turn+1)%4;
    }
    // winner
    var bw={seat:-1,power:-1};
    for(var w=0;w<play.length;w++){ var pw2=cardPow(play[w].card,trump,led); if(pw2>bw.power) bw={seat:play[w].seat,power:pw2}; }
    S.tricks[bw.seat]++; leader=bw.seat;
  }
  // score
  var mt=TEAM[maker]; var tt=[0,0]; for(var q=0;q<4;q++) tt[TEAM[q]]+=S.tricks[q];
  if(tt[0]+tt[1]!==5) return {pts:[0,0],err:'trick total '+(tt[0]+tt[1])};
  var pts=[0,0];
  if(tt[mt]>=3){ if(tt[mt]===5){ pts[mt]=alone?4:2; } else pts[mt]=1; }
  else pts[1-mt]=2;
  return {pts:pts, err:null};
}

function hStr(hand,trump){ var s=0; for(var i=0;i<hand.length;i++){ var c=hand[i]; if(isRight(c,trump))s+=15; else if(isLeft(c,trump))s+=13; else if(c.s===trump)s+=6+RANK_VAL[c.r]/5; else if(c.r==='A')s+=3; else s+=0.4; } return s; }

function legalSim(hand,trump,led){ if(led==null) return hand.map(function(_,i){return i;}); var f=[]; for(var i=0;i<hand.length;i++) if(effSuit(hand[i],trump)===led) f.push(i); return f.length?f:hand.map(function(_,i){return i;}); }
function aiPlaySim(hand,trump,led,play,p){
  var legal=legalSim(hand,trump,led);
  var bw={seat:-1,power:-1}; for(var w=0;w<play.length;w++){ var pw=cardPow(play[w].card,trump,led); if(pw>bw.power) bw={seat:play[w].seat,power:pw}; }
  var opts=legal.map(function(i){return {i:i,pw:cardPow(hand[i],trump,led)};});
  if(led==null){ opts.sort(function(a,b){return b.pw-a.pw;}); return opts[Math.floor(opts.length/2)].i; }
  var partnerWin = bw.seat>=0 && TEAM[bw.seat]===TEAM[p];
  var winners=opts.filter(function(o){return o.pw>bw.power;});
  if(partnerWin && bw.power>=500){ opts.sort(function(a,b){return a.pw-b.pw;}); return opts[0].i; }
  if(winners.length){ winners.sort(function(a,b){return a.pw-b.pw;}); return winners[0].i; }
  opts.sort(function(a,b){return a.pw-b.pw;}); return opts[0].i;
}
